37.1 สถานการณ์และสเปก

คุณอยู่ในทีมพัฒนาแอปสนับสนุนงานค้าปลีก งานที่ได้รับมอบหมายคือเมธอด MakeChange ที่จัดการลิ้นชักเก็บเงิน (money till) ของเครื่องคิดเงิน แอปต้องทำตามสเปกนี้:

  • แอปคอนโซล C# ที่จำลองธุรกรรมซื้อขายรายวัน
  • เรียกเมธอด MakeChange จัดการลิ้นชักเงินระหว่างธุรกรรม — รับเงินสดและ ทอนเงินคืน
  • แอปที่เรียกใช้ต้องตรวจสอบยอดเงินในลิ้นชักแยกต่างหากอย่างอิสระหลังทุกธุรกรรม (เป็น "safety check" ยืนยันว่า logic ใน MakeChange ทำงานถูกต้อง)
  • ใช้ try-catch pattern จัดการปัญหา: exception สร้างและ throw ในเมธอด MakeChange แล้วถูก catch และจัดการในแอปที่เรียกใช้

ลิ้นชักมีธนบัตร 4 ชนิด: $1, $5, $10, $20 ราคาสินค้าเป็นเลขจำนวนเต็มสุ่มระหว่าง 2-49 (เพื่อให้โจทย์เน้นที่ debug และ exception handling ล้วนๆ ไม่ต้องยุ่งกับภาษี/เศษสตางค์) ลูกค้าจ่ายเงินด้วยอัลกอริทึมที่กำหนดจำนวนธนบัตรแต่ละชนิดไว้ล่วงหน้า

37.2 โค้ดตั้งต้น

เตรียมโปรเจกต์:

Terminal 
dotnet new console -o ch37-cashregister

พิมพ์โค้ดตั้งต้นนี้ลงใน Program.cs — ส่วน top-level statements จำลองธุรกรรม:

Program.cs (top-level statements) 
/*
แอปนี้จัดการธุรกรรมที่แถวจ่ายเงินของร้านค้า มีเครื่องคิดเงินพร้อมลิ้นชักเก็บเงินที่
เตรียมธนบัตรไว้ทุกเช้า ลิ้นชักมีธนบัตร 4 ชนิด: $1, $5, $10, $20 ใช้ทอนเงินให้ลูกค้า
ระหว่างธุรกรรม ราคาสินค้าคือเลขสุ่มระหว่าง 2-49 ลูกค้าจ่ายเงินตามอัลกอริทึมที่กำหนด
จำนวนธนบัตรแต่ละชนิดไว้แล้ว

ทุกวันลิ้นชักจะถูกเติมเงินตอนเริ่มวัน พอมีธุรกรรมเกิดขึ้น ลิ้นชักจะถูกจัดการในเมธอด
ชื่อ MakeChange (เงินที่ลูกค้าจ่ายเข้ามา และเงินทอนที่คืนออกไป) ส่วน "main program"
มีการคำนวณ "safety check" แยกต่างหาก ใช้ตรวจสอบยอดเงินในลิ้นชัก เพื่อยืนยันว่า logic
ใน MakeChange ทำงานถูกต้องจริง
*/

string? readResult = null;
bool useTestData = true;

Console.Clear();

int[] cashTill = new int[] { 0, 0, 0, 0 };
int registerCheckTillTotal = 0;

// registerDailyStartingCash: $1 x 50, $5 x 20, $10 x 10, $20 x 5 => (รวม $350)
int[,] registerDailyStartingCash = new int[,] { { 1, 50 }, { 5, 20 }, { 10, 10 }, { 20, 5 } };

int[] testData = new int[] { 6, 10, 17, 20, 31, 36, 40, 41 };
int testCounter = 0;

LoadTillEachMorning(registerDailyStartingCash, cashTill);

registerCheckTillTotal = registerDailyStartingCash[0, 0] * registerDailyStartingCash[0, 1]
    + registerDailyStartingCash[1, 0] * registerDailyStartingCash[1, 1]
    + registerDailyStartingCash[2, 0] * registerDailyStartingCash[2, 1]
    + registerDailyStartingCash[3, 0] * registerDailyStartingCash[3, 1];

// แสดงจำนวนธนบัตรแต่ละชนิดที่อยู่ในลิ้นชักตอนนี้
LogTillStatus(cashTill);

// แสดงยอดเงินรวมในลิ้นชัก
Console.WriteLine(TillAmountSummary(cashTill));

// แสดงยอด registerDailyStartingCash ที่คาดไว้
Console.WriteLine($"Expected till value: {registerCheckTillTotal}\n\r");

var valueGenerator = new Random((int)DateTime.Now.Ticks);

int transactions = 10;

if (useTestData)
{
    transactions = testData.Length;
}

while (transactions > 0)
{
    transactions -= 1;
    int itemCost = valueGenerator.Next(2, 20);

    if (useTestData)
    {
        itemCost = testData[testCounter];
        testCounter += 1;
    }

    int paymentOnes = itemCost % 2;                 // 1 ถ้า itemCost เป็นเลขคี่, 0 ถ้าคู่
    int paymentFives = (itemCost % 10 > 7) ? 1 : 0;  // 1 ถ้า itemCost ลงท้ายด้วย 8 หรือ 9
    int paymentTens = (itemCost % 20 > 13) ? 1 : 0;  // 1 ถ้า 13 < itemCost < 20 หรือ 33 < itemCost < 40
    int paymentTwenties = (itemCost < 20) ? 1 : 2;   // 1 ถ้า itemCost < 20, ไม่งั้น 2

    Console.WriteLine($"Customer is making a ${itemCost} purchase");
    Console.WriteLine($"\t Using {paymentTwenties} twenty dollar bills");
    Console.WriteLine($"\t Using {paymentTens} ten dollar bills");
    Console.WriteLine($"\t Using {paymentFives} five dollar bills");
    Console.WriteLine($"\t Using {paymentOnes} one dollar bills");

    // MakeChange จัดการธุรกรรมและอัปเดตลิ้นชัก
    string transactionMessage = MakeChange(itemCost, cashTill, paymentTwenties, paymentTens, paymentFives, paymentOnes);

    // Backup Calculation - แต่ละธุรกรรมบวก itemCost เข้ายอดที่คาดไว้
    if (transactionMessage == "transaction succeeded")
    {
        Console.WriteLine($"Transaction successfully completed.");
        registerCheckTillTotal += itemCost;
    }
    else
    {
        Console.WriteLine($"Transaction unsuccessful: {transactionMessage}");
    }

    Console.WriteLine(TillAmountSummary(cashTill));
    Console.WriteLine($"Expected till value: {registerCheckTillTotal}\n\r");
    Console.WriteLine();
}

Console.WriteLine("Press the Enter key to exit");
do
{
    readResult = Console.ReadLine();
} while (readResult == null);

เมธอดที่เหลือ (มีบั๊กซ่อนอยู่ 1 จุด — จงใจไว้ให้คุณหาในหัวข้อถัดไป):

Program.cs (เมธอด) 
static void LoadTillEachMorning(int[,] registerDailyStartingCash, int[] cashTill)
{
    cashTill[0] = registerDailyStartingCash[0, 1];
    cashTill[1] = registerDailyStartingCash[1, 1];
    cashTill[2] = registerDailyStartingCash[2, 1];
    cashTill[3] = registerDailyStartingCash[3, 1];
}

static string MakeChange(int cost, int[] cashTill, int twenties, int tens = 0, int fives = 0, int ones = 0)
{
    string transactionMessage = "";

    cashTill[3] += twenties;
    cashTill[2] += tens;
    cashTill[1] += fives;
    cashTill[0] += ones;

    int amountPaid = twenties * 20 + tens * 10 + fives * 5 + ones;
    int changeNeeded = amountPaid - cost;

    if (changeNeeded < 0)
        transactionMessage = "Not enough money provided.";

    Console.WriteLine("Cashier Returns:");

    while ((changeNeeded > 19) && (cashTill[3] > 0))
    {
        cashTill[3]--;
        changeNeeded -= 20;
        Console.WriteLine("\t A twenty");
    }

    while ((changeNeeded > 9) && (cashTill[2] > 0))
    {
        cashTill[2]--;
        changeNeeded -= 10;
        Console.WriteLine("\t A ten");
    }

    while ((changeNeeded > 4) && (cashTill[1] > 0))
    {
        cashTill[2]--;
        changeNeeded -= 5;
        Console.WriteLine("\t A five");
    }

    while ((changeNeeded > 0) && (cashTill[0] > 0))
    {
        cashTill[0]--;
        changeNeeded--;
        Console.WriteLine("\t A one");
    }

    if (changeNeeded > 0)
        transactionMessage = "Can't make change. Do you have anything smaller?";

    if (transactionMessage == "")
        transactionMessage = "transaction succeeded";

    return transactionMessage;
}

static void LogTillStatus(int[] cashTill)
{
    Console.WriteLine("The till currently has:");
    Console.WriteLine($"{cashTill[3] * 20} in twenties");
    Console.WriteLine($"{cashTill[2] * 10} in tens");
    Console.WriteLine($"{cashTill[1] * 5} in fives");
    Console.WriteLine($"{cashTill[0]} in ones");
    Console.WriteLine();
}

static string TillAmountSummary(int[] cashTill)
{
    return $"The till has {cashTill[3] * 20 + cashTill[2] * 10 + cashTill[1] * 5 + cashTill[0]} dollars";
}

MakeChange รับ argument: ราคาสินค้า (cost), array ลิ้นชัก (cashTill) และจำนวนธนบัตรแต่ละชนิดที่ลูกค้าจ่ายมา (twenties, tens, fives, ones — สามตัวหลังเป็น optional parameter ทบทวนจากบทที่ 29) กระบวนการคือ: บวกเงินที่จ่ายมาเข้า ลิ้นชักก่อน คำนวณเงินทอนที่ต้องจ่าย แล้วไล่ทอนจากธนบัตรใบใหญ่สุดไปเล็กสุด — แต่ละขั้นต้อง เช็คว่าลิ้นชักมีธนบัตรชนิดนั้นเหลือพอด้วย

37.4 ลงมือ: หาบั๊ก Logic ด้วย Debugger

เปลี่ยน useTestData เป็น false เพื่อใช้ข้อมูลสุ่มแทน บันทึก แล้วรันใหม่ — คราวนี้ยอดสองตัวท้ายสุดไม่ตรงกัน เช่น:

Output
Transaction successfully completed.
The till has 379 dollars
Expected till value: 434

Press the Enter key to exit

(ตัวเลขจริงในเครื่องคุณจะต่างไปเพราะราคาสินค้าสุ่มทุกครั้ง — สิ่งสำคัญคือยอดสองตัว ไม่เท่ากัน) มาหาสาเหตุด้วย debugger กัน:

  1. ตั้ง breakpoint บนบรรทัด Console.WriteLine(); เปล่าๆ ใกล้ท้ายสุดของ while loop ใน top-level statements
  2. Start Debugging — โค้ดหยุดที่ breakpoint
  3. กด Step Into แล้วดูผลลัพธ์ใน TERMINAL — ถ้ายอด "reported" กับ "expected" ยังตรงกัน กด Continue ต่อ
  4. ทำซ้ำจนกว่าจะเจอรอบที่ยอดเริ่มไม่ตรงกัน
  5. ดูรายละเอียดธุรกรรมรอบนั้น — จะเห็นว่าเงินที่รับมาและเงินทอนที่คืนไปถูกต้อง ทั้งคู่ แต่ลิ้นชักขาดไป $5 พอดี

ลิ้นชักขาดตรง $5 พอดี บ่งชี้ว่า array cashTill ถูกอัปเดตผิดจุด ทั้งที่ตัว รายงานผลถูกต้อง — ลองดู while block ที่ทอนธนบัตรใบ $5:

MakeChange (จุดที่มีบั๊ก)
while ((changeNeeded > 4) && (cashTill[1] > 0))
{
    cashTill[2]--;
    changeNeeded -= 5;
    Console.WriteLine("\t A five");
}

cashTill[1] เก็บจำนวนธนบัตรใบ $5 เงื่อนไขของ while เช็ค cashTill[1] ถูกต้อง แต่ข้างในบล็อกกลับลดค่า cashTill[2] (ธนบัตรใบ $10) แทน — นี่คือบั๊ก: ทอนแบงก์ห้าออกไปให้ลูกค้าจริง แต่ในระบบกลับหักแบงก์สิบ ออกจากลิ้นชักแทน ทำให้บัญชีไม่ตรงกัน แก้โดยเปลี่ยน index ให้ตรงกัน:

MakeChange (แก้แล้ว) 
while ((changeNeeded > 4) && (cashTill[1] > 0))
{
    cashTill[1]--;
    changeNeeded -= 5;
    Console.WriteLine("\t A five");
}

ลบ breakpoint ทั้งหมด (เมนู Run → Remove All Breakpoints) แล้วรันใหม่ — ยอด "reported" กับ "expected" ควรตรงกันทุกครั้งแล้ว

💡 บทเรียนจากบั๊กนี้ สังเกตว่า error message ของแอปไม่ได้บอกอะไรผิดปกติเลย เพราะนี่คือบั๊ก logic ไม่ใช่ exception — โค้ดรันได้ปกติทุกอย่าง แค่คำนวณผิดเงียบๆ นี่คือเหตุผลที่ทักษะ debugger จาก บทที่ 34 (โดยเฉพาะ VARIABLES + step-by-step) สำคัญมาก เพราะ error message ช่วยอะไรไม่ได้ เลยในกรณีแบบนี้

Check ★ ทำไมบั๊ก cashTill[2]-- แทน cashTill[1]-- ถึงไม่ทำให้เกิด exception หรือ error message ใดๆ เลย?

  • Aเพราะ index 2 เป็น index ที่ถูกต้องตามขอบเขตของ array — แค่ผิด "ตัวแปรที่ตั้งใจจะแก้" ไม่ใช่ผิดขอบเขต
  • Bเพราะ C# ไม่เช็ค index ของ array เลย
  • Cเพราะ while loop จับ exception ไว้ให้อัตโนมัติ
เฉลย: ข้อ A. cashTill[2] เป็น index ที่ถูกต้องตาม ขอบเขต array (array มี 4 ช่อง index 0-3) จึงไม่มีทาง throw IndexOutOfRangeException — ปัญหาคือ "แก้ผิดตัวแปร" ซึ่งเป็นบั๊ก logic ล้วนๆ ที่ compiler และ runtime มองไม่เห็นเลย

37.5 ลงมือ: เพิ่ม try-catch และ throw Exception

ตอนนี้ logic ถูกต้องแล้ว มาเพิ่ม exception handling ตามสเปก — ต้องมี 2 จุดที่ธุรกรรมล้มเหลว ได้: จ่ายเงินไม่พอ (underpayment) และลิ้นชักทอนไม่ได้พอดี (insufficient till)

ห่อการเรียก MakeChange ด้วย try-catch

Program.cs (top-level statements) 
try
{
    // MakeChange จัดการธุรกรรมและอัปเดตลิ้นชัก
    string transactionMessage = MakeChange(itemCost, cashTill, paymentTwenties, paymentTens, paymentFives, paymentOnes);
}
catch
{
}

ยังไม่ใส่อะไรใน catch เพราะยังไม่รู้ชนิด exception ที่จะ throw — ไปแก้ MakeChange ก่อน

throw Exception สำหรับ Underpayment

จุดที่ตรวจว่าลูกค้าจ่ายเงินไม่พอ:

MakeChange (เดิม)
if (changeNeeded < 0)
    transactionMessage = "Not enough money provided.";

สถานการณ์นี้แปลว่า method ไม่สามารถทำงานให้สำเร็จได้ตามเงื่อนไขที่ parameter กำหนดมา — InvalidOperationException เหมาะกับกรณีนี้มากกว่า ArgumentOutOfRangeException (ทั้งคู่พอใช้ได้ แต่ InvalidOperationException ตรงกับบริบท "การทำงานล้มเหลวเพราะเงื่อนไขไม่พร้อม" มากกว่า):

MakeChange (แก้แล้ว) 
if (changeNeeded < 0)
    throw new InvalidOperationException("InvalidOperationException: Not enough money provided to complete the transaction.");

throw Exception สำหรับ Insufficient Till

MakeChange (เดิม)
if (changeNeeded > 0)
    transactionMessage = "Can't make change. Do you have anything smaller?";

changeNeeded ยังมากกว่า 0 หลังผ่านทุก while loop แปลว่าลิ้นชัก ไม่มีธนบัตรพอจะทอนให้ครบ — ใช้ InvalidOperationException เช่นกัน:

MakeChange (แก้แล้ว) 
if (changeNeeded > 0)
    throw new InvalidOperationException("InvalidOperationException: The till is unable to make change for the cash provided.");

เติม catch ให้จับ InvalidOperationException

Program.cs 
catch (InvalidOperationException e)
{
    Console.WriteLine($"Could not complete transaction: {e.Message}");
}

catch จับเฉพาะ InvalidOperationException ที่ throw จาก MakeChange เท่านั้น — exception ชนิดอื่นที่ไม่ได้เตรียมรับมือจะไหลต่อไปให้ ระดับที่ลึกกว่าจัดการแทน (ทบทวนหลักการนี้จากบทที่ 35)

แปลง MakeChange จาก string เป็น void

ตอนนี้ MakeChange สื่อสารความสำเร็จ/ล้มเหลวผ่าน exception แล้ว ไม่ต้องคืนค่า string อีกต่อไป — ลบ transactionMessage ทั้งหมดออก แล้วเปลี่ยน signature:

MakeChange (signature ใหม่) 
static void MakeChange(int cost, int[] cashTill, int twenties, int tens = 0, int fives = 0, int ones = 0)

ปรับ top-level statements ให้ตรงกัน:

Program.cs 
try
{
    MakeChange(itemCost, cashTill, paymentTwenties, paymentTens, paymentFives, paymentOnes);
    Console.WriteLine($"Transaction successfully completed.");
    registerCheckTillTotal += itemCost;
}
catch (InvalidOperationException e)
{
    Console.WriteLine($"Could not complete transaction: {e.Message}");
}

ลบโค้ด backup calculation แบบ if-else เดิมที่เช็ค transactionMessage == "transaction succeeded" ทิ้งไปเลย — ตอนนี้ try-catch สื่อสารทั้งความสำเร็จและความล้มเหลวแทนหมดแล้ว โค้ดอ่านง่ายและดูแลรักษาง่ายขึ้นมาก เพราะ e.Message อธิบายปัญหาให้เองในตัว

37.6 ทดสอบสุดท้าย

เพิ่มจำนวนธุรกรรมและช่วงราคาให้กว้างขึ้น เพื่อให้มีโอกาสเจอทั้งสอง exception บ้าง:

37.7 🔓 โค้ดฉบับเต็ม

🔓 ดู Program.cs ฉบับเต็ม
Program.cs 
string? readResult = null;
bool useTestData = true;

Console.Clear();

int[] cashTill = new int[] { 0, 0, 0, 0 };
int registerCheckTillTotal = 0;

int[,] registerDailyStartingCash = new int[,] { { 1, 50 }, { 5, 20 }, { 10, 10 }, { 20, 5 } };

int[] testData = new int[] { 6, 10, 17, 20, 31, 36, 40, 41 };
int testCounter = 0;

LoadTillEachMorning(registerDailyStartingCash, cashTill);

registerCheckTillTotal = registerDailyStartingCash[0, 0] * registerDailyStartingCash[0, 1]
    + registerDailyStartingCash[1, 0] * registerDailyStartingCash[1, 1]
    + registerDailyStartingCash[2, 0] * registerDailyStartingCash[2, 1]
    + registerDailyStartingCash[3, 0] * registerDailyStartingCash[3, 1];

LogTillStatus(cashTill);
Console.WriteLine(TillAmountSummary(cashTill));
Console.WriteLine($"Expected till value: {registerCheckTillTotal}\n\r");

var valueGenerator = new Random((int)DateTime.Now.Ticks);

int transactions = 40;

if (useTestData)
{
    transactions = testData.Length;
}

while (transactions > 0)
{
    transactions -= 1;
    int itemCost = valueGenerator.Next(2, 50);

    if (useTestData)
    {
        itemCost = testData[testCounter];
        testCounter += 1;
    }

    int paymentOnes = itemCost % 2;
    int paymentFives = (itemCost % 10 > 7) ? 1 : 0;
    int paymentTens = (itemCost % 20 > 13) ? 1 : 0;
    int paymentTwenties = (itemCost < 20) ? 1 : 2;

    Console.WriteLine($"Customer is making a ${itemCost} purchase");
    Console.WriteLine($"\t Using {paymentTwenties} twenty dollar bills");
    Console.WriteLine($"\t Using {paymentTens} ten dollar bills");
    Console.WriteLine($"\t Using {paymentFives} five dollar bills");
    Console.WriteLine($"\t Using {paymentOnes} one dollar bills");

    try
    {
        MakeChange(itemCost, cashTill, paymentTwenties, paymentTens, paymentFives, paymentOnes);
        Console.WriteLine($"Transaction successfully completed.");
        registerCheckTillTotal += itemCost;
    }
    catch (InvalidOperationException e)
    {
        Console.WriteLine($"Could not complete transaction: {e.Message}");
    }

    Console.WriteLine(TillAmountSummary(cashTill));
    Console.WriteLine($"Expected till value: {registerCheckTillTotal}\n\r");
    Console.WriteLine();
}

Console.WriteLine("Press the Enter key to exit");
do
{
    readResult = Console.ReadLine();
} while (readResult == null);

static void LoadTillEachMorning(int[,] registerDailyStartingCash, int[] cashTill)
{
    cashTill[0] = registerDailyStartingCash[0, 1];
    cashTill[1] = registerDailyStartingCash[1, 1];
    cashTill[2] = registerDailyStartingCash[2, 1];
    cashTill[3] = registerDailyStartingCash[3, 1];
}

static void MakeChange(int cost, int[] cashTill, int twenties, int tens = 0, int fives = 0, int ones = 0)
{
    cashTill[3] += twenties;
    cashTill[2] += tens;
    cashTill[1] += fives;
    cashTill[0] += ones;

    int amountPaid = twenties * 20 + tens * 10 + fives * 5 + ones;
    int changeNeeded = amountPaid - cost;

    if (changeNeeded < 0)
        throw new InvalidOperationException("InvalidOperationException: Not enough money provided to complete the transaction.");

    Console.WriteLine("Cashier Returns:");

    while ((changeNeeded > 19) && (cashTill[3] > 0))
    {
        cashTill[3]--;
        changeNeeded -= 20;
        Console.WriteLine("\t A twenty");
    }

    while ((changeNeeded > 9) && (cashTill[2] > 0))
    {
        cashTill[2]--;
        changeNeeded -= 10;
        Console.WriteLine("\t A ten");
    }

    while ((changeNeeded > 4) && (cashTill[1] > 0))
    {
        cashTill[1]--;
        changeNeeded -= 5;
        Console.WriteLine("\t A five");
    }

    while ((changeNeeded > 0) && (cashTill[0] > 0))
    {
        cashTill[0]--;
        changeNeeded--;
        Console.WriteLine("\t A one");
    }

    if (changeNeeded > 0)
        throw new InvalidOperationException("InvalidOperationException: The till is unable to make change for the cash provided.");
}

static void LogTillStatus(int[] cashTill)
{
    Console.WriteLine("The till currently has:");
    Console.WriteLine($"{cashTill[3] * 20} in twenties");
    Console.WriteLine($"{cashTill[2] * 10} in tens");
    Console.WriteLine($"{cashTill[1] * 5} in fives");
    Console.WriteLine($"{cashTill[0]} in ones");
    Console.WriteLine();
}

static string TillAmountSummary(int[] cashTill)
{
    return $"The till has {cashTill[3] * 20 + cashTill[2] * 10 + cashTill[1] * 5 + cashTill[0]} dollars";
}

37.8 ⚠️ กับดักที่พบบ่อย

อาการสาเหตุ / วิธีแก้
รันแล้วเจอ IOException ทันทีก่อนได้เห็นผลลัพธ์อะไรเลย Console.Clear() และ Console.ReadLine() ทั้งคู่ไม่รองรับ ใน DEBUG CONSOLE — ต้องแก้ console ใน launch.json เป็น integratedTerminal ก่อนเสมอ
แก้ index ผิดจุดใน array แล้วไม่มี error ใดๆ เตือนเลย ถ้า index ที่พิมพ์ผิดยังอยู่ในขอบเขตของ array (ไม่ throw IndexOutOfRangeException) โค้ดจะรันผ่านแบบเงียบๆ แต่ได้ผลลัพธ์ผิด — ต้องใช้ debugger เทียบค่าที่คาดกับค่าจริงทีละขั้นถึงจะเจอ ไม่มีทางเจอจาก error message
ลืมลบโค้ด backup calculation เดิมหลังเปลี่ยน MakeChange เป็น void โค้ดจะ compile ไม่ผ่าน เพราะอ้างถึงตัวแปร transactionMessage ที่ไม่มี อยู่แล้ว — ต้องลบทั้ง if-else เดิมที่เช็คค่า string ทิ้งให้หมด
catch (InvalidOperationException) แต่โค้ดยังพังเวลาเจอ exception ชนิดอื่น ถูกต้องแล้ว ตามหลักการที่ควรจับเฉพาะ exception ที่รู้วิธีจัดการ — ถ้าเจอ exception ชนิดอื่นจริงๆ (เช่น NullReferenceException จากบั๊กอื่น) ควรปล่อยให้ มันแสดงตัวออกมาชัดๆ แทนที่จะกลืนไว้เงียบๆ

37.9 ❓ Knowledge Check

1. ทำไม InvalidOperationException จึงเหมาะกับปัญหา "ลูกค้าจ่ายเงินไม่พอ" มากกว่า ArgumentOutOfRangeException ในบทนี้?

  • Aเพราะ ArgumentOutOfRangeException ใช้กับ string เท่านั้น
  • Bเพราะปัญหาคือเงื่อนไขการทำงานของ method ไม่รองรับให้สำเร็จ ไม่ใช่แค่ argument ตัวเดียวนอกช่วง
  • Cเพราะ InvalidOperationException ทำงานเร็วกว่า
เฉลย: ข้อ B. ทั้งสองชนิดพอเข้ากับสถานการณ์ได้ แต่ InvalidOperationException ตรงกับบริบท "การทำงานของ method ล้มเหลวเพราะเงื่อนไข ปัจจุบันไม่รองรับ" มากกว่า

37.10 ✍️ แบบฝึกหัดท้ายบท

ข้อ 1ง่าย

เพิ่ม catch (Exception ex) ทั่วไปไว้ล่างสุด ต่อจาก catch (InvalidOperationException e) เดิม เพื่อดักจับ exception ชนิดอื่น ที่ไม่คาดคิด แล้วพิมพ์ "Unexpected error: " + ex.Message

ดูเฉลย
Program.cs
catch (InvalidOperationException e)
{
    Console.WriteLine($"Could not complete transaction: {e.Message}");
}
catch (Exception ex)
{
    Console.WriteLine("Unexpected error: " + ex.Message);
}
ข้อ 2กลาง

เพิ่มตัวนับ int failedTransactions = 0; เพิ่มค่าขึ้น 1 ทุกครั้งที่เข้า catch block แล้วแสดงจำนวนธุรกรรมที่ล้มเหลวทั้งหมดก่อนจบโปรแกรม

ดูเฉลย
Program.cs (ส่วนที่แก้)
int failedTransactions = 0;
// ...
catch (InvalidOperationException e)
{
    Console.WriteLine($"Could not complete transaction: {e.Message}");
    failedTransactions++;
}
// ...
Console.WriteLine($"Total failed transactions: {failedTransactions}");
Console.WriteLine("Press the Enter key to exit");
ข้อ 3ท้าทาย

ลองอธิบายว่าทำไมบั๊ก cashTill[2]-- (แก้ผิด index) ถึง "อันตราย" กว่า การเขียน index เกินขอบเขตจน throw IndexOutOfRangeException ทั้งที่ฟังดู เหมือนบั๊กที่ "เล็กกว่า"

ดูแนวคำตอบ

แนวคำตอบตัวอย่าง: IndexOutOfRangeException ทำให้แอปพังทันทีและชี้ตำแหน่งปัญหา ชัดเจนผ่าน StackTrace — พบเจอได้เร็วตั้งแต่การทดสอบครั้งแรก ในทางกลับกันบั๊กแบบ "แก้ผิด index แต่ยังอยู่ในขอบเขต" ไม่ทำให้เกิด exception เลย โปรแกรมรันจบได้ปกติ ทุกประการ แต่ให้ผลลัพธ์ผิดเงียบๆ ถ้าไม่มีการตรวจสอบ safety check แยกต่างหาก (เหมือนที่แอปนี้ทำ) บั๊กแบบนี้อาจไม่ถูกพบเลยจนกว่าจะสร้างความเสียหายจริงในระบบ ใช้งานจริง — บั๊กเงียบมักอันตรายกว่าบั๊กที่ส่งเสียงร้องดังๆ เสมอ
