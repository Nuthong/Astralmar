  • เขียนแอปที่ catch exception ของตัวเอง แล้วใช้ข้อมูลใน property จัดการปัญหาได้
💡 อีกฝั่งหนึ่งของเหรียญ บทที่ 35 คุณเรียน "จับ" (catch) exception ที่คนอื่นโยนมา — บทนี้คุณจะเป็นฝ่าย "โยน" (throw) exception ของคุณเองเมื่อโค้ดของคุณตรวจพบปัญหาที่แก้ต่อไม่ได้

36.1 สร้าง Exception Object ของคุณเอง

.NET มีลำดับชั้นคลาส exception ที่สืบทอดมาจาก System.Exception แอป C# สร้างและ throw exception ชนิดไหนก็ได้ในลำดับชั้นนี้ นักพัฒนายังปรับแต่ง exception object ด้วยข้อมูลเฉพาะแอปได้ โดยกำหนดค่า property ของมัน

ชนิด exception ที่คุณสร้างควรตรงกับปัญหาที่เกิดขึ้นให้ใกล้เคียงที่สุด ตัวอย่าง exception ที่พบบ่อยเวลาสร้างเอง:

Exceptionใช้เมื่อ
ArgumentException / ArgumentNullException method หรือ constructor ถูกเรียกด้วย argument ที่ไม่ถูกต้องหรือเป็น null
InvalidOperationException เงื่อนไขการทำงานปัจจุบันของ method ไม่รองรับให้เรียกสำเร็จได้
NotSupportedExceptionการทำงานหรือฟีเจอร์นั้นไม่รองรับ
IOExceptionการอ่าน/เขียนไฟล์ล้มเหลว
FormatExceptionรูปแบบของ string หรือข้อมูลไม่ถูกต้อง

ใช้ new สร้าง instance ของ exception เหมือนสร้าง object ทั่วไป:

ตัวอย่าง
ArgumentException invalidArgumentException = new ArgumentException();

36.2 ปรับแต่งและ throw Exception

ขั้นตอนการ throw exception คือ: สร้าง instance ของคลาสที่สืบทอดจาก exception → ปรับแต่ง property (ถ้าต้องการ) → ใช้ throw โยนมันออกไป การใส่ข้อความบริบทลงใน exception ก่อน throw ช่วยได้มาก เช่น:

ตัวอย่าง
ArgumentException invalidArgumentException = new ArgumentException(
    "ArgumentException: เมธอด 'GraphData' ได้รับข้อมูลนอกช่วงที่กำหนดไว้");
throw invalidArgumentException;
ℹ️ Message เป็น readonly property Message ของ exception เป็น readonly เขียนทับไม่ได้หลังสร้างแล้ว ต้องกำหนดข้อความ custom ตอนสร้าง object เท่านั้น (ผ่าน constructor)

สร้าง exception object แล้ว throw ทันทีในบรรทัดเดียวก็ได้เช่นกัน:

ตัวอย่าง
throw new FormatException(
    "FormatException: การคำนวณในกระบวนการ XYZ ถูกยกเลิกเพราะข้อมูลผิดรูปแบบ");

ข้อควรระวังตอนใส่ข้อความลงใน exception: Message ควรอธิบายสาเหตุ แต่ห้าม ใส่ข้อมูลอ่อนไหวหรือเกี่ยวกับความปลอดภัยลงไป ส่วน StackTrace ใช้ตามหา ต้นตอของ exception ได้ — CLR สร้าง StackTrace ให้อัตโนมัติจากจุดที่ throw ถูกเรียก ดังนั้น exception ต้องถูก throw จากจุดที่ต้องการให้ stack trace เริ่มนับจริงๆ

Check ★ ทำไมต้องตั้งค่า Message ตอน "สร้าง" exception object เท่านั้น ไม่ใช่ตอนหลัง?

  • Aเพราะ C# ไม่รองรับการเขียนโค้ดหลายบรรทัดก่อน throw
  • Bเพราะ property Message เป็น readonly แก้ไขทีหลังไม่ได้
  • Cเพราะ throw ต้องอยู่บรรทัดแรกของ method เสมอ
เฉลย: ข้อ B. Message กำหนดได้ผ่าน constructor เท่านั้น จะมา เขียนทับทีหลังไม่ได้

36.3 ลงมือ: throw Exception เมื่อ input ไม่ถูกต้อง

เตรียมโปรเจกต์

Terminal 
dotnet new console -o ch36-throw
⚠️ อย่าลืมแก้ launch.json โปรเจกต์นี้ใช้ Console.ReadLine() รับค่าจากผู้ใช้ — ต้องแก้ attribute console ใน launch.json เป็น integratedTerminal ก่อน debug (ทบทวนกับดักนี้จากบทที่ 34)

พิมพ์โค้ดนี้ลงใน Program.cs — เมธอดคำนวณค่าเฉลี่ยของเลขคู่ระหว่างขอบเขต ที่กำหนด:

Program.cs 
Console.Write("กรอกขอบเขตล่าง: ");
int lowerBound = int.Parse(Console.ReadLine());

Console.Write("กรอกขอบเขตบน: ");
int upperBound = int.Parse(Console.ReadLine());

decimal averageValue = AverageOfEvenNumbers(lowerBound, upperBound);
Console.WriteLine($"ค่าเฉลี่ยเลขคู่ระหว่าง {lowerBound} ถึง {upperBound} คือ {averageValue}");

static decimal AverageOfEvenNumbers(int lowerBound, int upperBound)
{
    int sum = 0;
    int count = 0;

    for (int i = lowerBound; i <= upperBound; i++)
    {
        if (i % 2 == 0)
        {
            sum += i;
            count++;
        }
    }

    return (decimal)sum / count;
}

ถ้าผู้ใช้กรอก lowerBound มากกว่าหรือเท่ากับ upperBound (เช่น 3 กับ 3) จะไม่มีเลขคู่ให้ นับเลย count จะเป็น 0 และเกิด DivideByZeroException ตอนคำนวณ ค่าเฉลี่ย — วิธีที่ดีกว่าคือตรวจสอบ parameter ก่อนเริ่มคำนวณ แล้ว throw exception ทันทีถ้าค่าที่ได้มาไม่สมเหตุสมผล แทนที่จะปล่อยให้คำนวณไปเจอปัญหาเอาทีหลัง

exception ชนิดที่เหมาะกับปัญหานี้มีสองตัวให้พิจารณา: ArgumentOutOfRangeException (ขอบเขตของ argument ที่รับมาไม่ถูกต้อง) กับ InvalidOperationException (เงื่อนไขการทำงานของ method ไม่รองรับ) — เมื่อทั้งคู่เข้ากับปัญหาพอๆ กัน หลักการคือ เลือกตัวที่ขอบเขตแคบกว่า — ArgumentOutOfRangeException ผูกกับ argument ที่รับมาโดยตรง จึงเหมาะกว่า

เพิ่มการตรวจสอบไว้ต้นเมธอด:

Program.cs (เพิ่มการตรวจสอบ) 
static decimal AverageOfEvenNumbers(int lowerBound, int upperBound)
{
    if (lowerBound >= upperBound)
    {
        throw new ArgumentOutOfRangeException("upperBound",
            "ArgumentOutOfRangeException: ขอบเขตบนต้องมากกว่าขอบเขตล่าง");
    }

    int sum = 0;
    int count = 0;

    for (int i = lowerBound; i <= upperBound; i++)
    {
        if (i % 2 == 0)
        {
            sum += i;
            count++;
        }
    }

    return (decimal)sum / count;
}

ArgumentOutOfRangeException constructor แบบนี้รับสอง argument: ชื่อ parameter ที่มีปัญหา ("upperBound") และข้อความอธิบาย — ฝั่งเรียกใช้ควรดัก ปัญหานี้ด้วย try-catch แล้วให้โอกาสผู้ใช้กรอกค่าใหม่ผ่าน loop do-while:

Program.cs (ฉบับสมบูรณ์) 
Console.Write("กรอกขอบเขตล่าง: ");
int lowerBound = int.Parse(Console.ReadLine());

Console.Write("กรอกขอบเขตบน: ");
int upperBound = int.Parse(Console.ReadLine());

decimal averageValue = 0;
bool exit = false;

do
{
    try
    {
        averageValue = AverageOfEvenNumbers(lowerBound, upperBound);
        Console.WriteLine($"ค่าเฉลี่ยเลขคู่ระหว่าง {lowerBound} ถึง {upperBound} คือ {averageValue}");
        exit = true;
    }
    catch (ArgumentOutOfRangeException ex)
    {
        Console.WriteLine("เกิดข้อผิดพลาด");
        Console.WriteLine(ex.Message);
        Console.Write("กรอกขอบเขตบนใหม่ (หรือพิมพ์ exit เพื่อออก): ");
        string? userResponse = Console.ReadLine();
        if (userResponse.ToLower().Contains("exit"))
        {
            exit = true;
        }
        else
        {
            upperBound = int.Parse(userResponse);
        }
    }
} while (exit == false);

static decimal AverageOfEvenNumbers(int lowerBound, int upperBound)
{
    if (lowerBound >= upperBound)
    {
        throw new ArgumentOutOfRangeException("upperBound",
            "ArgumentOutOfRangeException: ขอบเขตบนต้องมากกว่าขอบเขตล่าง");
    }

    int sum = 0;
    int count = 0;

    for (int i = lowerBound; i <= upperBound; i++)
    {
        if (i % 2 == 0)
        {
            sum += i;
            count++;
        }
    }

    return (decimal)sum / count;
}

สังเกตว่าใช้ do-while (ทบทวนจากบทที่ 18) เพราะ ต้องการเรียก AverageOfEvenNumbers อย่างน้อยหนึ่งครั้งก่อนตรวจเงื่อนไข exit — ตัวแปร exit เป็นตัวควบคุมว่าจะวนต่อหรือไม่ ตั้งเป็น true ทั้งตอนสำเร็จและตอนผู้ใช้เลือกพิมพ์ "exit"

36.4 Re-throw Exception

นอกจาก throw exception ใหม่ คุณยัง throw ซ้ำ (re-throw) exception เดิม จากใน catch block ได้ — กรณีนี้ throw ไม่มี operand ตามหลัง:

ตัวอย่าง re-throw
catch (Exception ex)
{
    // จัดการหรือจัดการบางส่วน
    // ...

    throw; // โยน exception object เดิมต่อให้ระดับที่ลึกกว่าจัดการต่อ
}

การ re-throw แบบนี้ใช้ exception object ตัวเดิม จึงไม่เสียข้อมูลอะไรไป ถ้าอยาก สร้าง exception ใหม่ที่ "ห่อ" exception เดิมไว้ข้างใน ส่งตัวเดิมเป็น argument ให้ constructor ของตัวใหม่ได้:

ตัวอย่าง wrap exception
catch (Exception ex)
{
    // จัดการหรือจัดการบางส่วน
    // ...

    throw new ApplicationException("เกิดข้อผิดพลาด", ex); // ex กลายเป็น InnerException
}

ลองดูตัวอย่างที่ผสมทั้งสองแบบ: BusinessProcess1 ก่อ exception ได้สองชนิด (format ผิดหรือหารด้วยศูนย์) ส่วน OperatingProcedure1 ที่เรียกมันจัดการ FormatException ได้เต็มที่ แต่จัดการ DivideByZeroException ได้แค่บางส่วน แล้วต้อง re-throw ต่อให้ top-level statements จัดการต่อ:

Program.cs 
try
{
    OperatingProcedure1();
}
catch (Exception ex)
{
    Console.WriteLine(ex.Message);
    Console.WriteLine("ปิดโปรแกรม");
}

static void OperatingProcedure1()
{
    string[][] userEnteredValues = new string[][]
    {
        new string[] { "1", "two", "3" },
        new string[] { "0", "1", "2" }
    };

    foreach (string[] userEntries in userEnteredValues)
    {
        try
        {
            BusinessProcess1(userEntries);
        }
        catch (FormatException ex)
        {
            Console.WriteLine(ex.Message);
            Console.WriteLine("แก้ไขเรียบร้อยใน OperatingProcedure1");
        }
        catch (DivideByZeroException ex)
        {
            Console.WriteLine(ex.Message);
            Console.WriteLine("แก้ไขได้บางส่วนใน OperatingProcedure1 - ต้องจัดการต่อ");
            throw; // re-throw ตัวเดิมต่อให้ top-level statements
        }
    }
}

static void BusinessProcess1(string[] userEntries)
{
    foreach (string userValue in userEntries)
    {
        int valueEntered = int.Parse(userValue);

        checked
        {
            int calculatedValue = 4 / valueEntered;
        }
    }
}
Output
System.FormatException: The input string 'two' was not in a correct format.
แก้ไขเรียบร้อยใน OperatingProcedure1
System.DivideByZeroException: Attempted to divide by zero.
แก้ไขได้บางส่วนใน OperatingProcedure1 - ต้องจัดการต่อ
System.DivideByZeroException: Attempted to divide by zero.
ปิดโปรแกรม

สังเกตว่า exception ตัวเดียวกัน (DivideByZeroException) ถูกพิมพ์ข้อความ ออกมาสองครั้ง — ครั้งแรกใน OperatingProcedure1 (จัดการบางส่วน) ครั้งที่สอง ใน top-level statements (จัดการต่อจนจบ) เพราะ throw; ส่งต่อ exception object ตัวเดิมขึ้นไปให้ระดับที่สูงกว่าอีกที

36.5 ⚠️ สิ่งที่ควรเลี่ยงตอน throw Exception

  • อย่าใช้ exception เปลี่ยนทิศทางการทำงานของโปรแกรมแบบปกติ — ใช้ exception รายงานและ จัดการเงื่อนไข "ผิดปกติ" เท่านั้น
  • อย่าคืน exception เป็นค่า return หรือ parameter แทนการ throw จริง
  • อย่า throw System.Exception, System.SystemException, System.NullReferenceException หรือ System.IndexOutOfRangeException โดยตั้งใจจากโค้ดของคุณเอง — พวกนี้ควรเป็น exception ที่ runtime สร้างเอง ไม่ใช่ที่ คุณสร้างเอง
  • อย่าสร้างเงื่อนไขที่ throw exception ได้เฉพาะตอน debug mode แต่ release mode ไม่ throw — ถ้าต้องการเช็คเงื่อนไขที่ "ไม่ควรเกิดขึ้นเลย" ระหว่างพัฒนา ให้ใช้ Debug.Assert() แทน (ใช้เฉพาะตอน debug build เท่านั้น ไม่ใช่แทนที่ exception handling ของ flow ปกติ)

Check ★ ข้อใดคือแนวทางที่ควรเลี่ยงตอน throw exception?

  • Athrow exception ชนิดที่เจาะจงกับปัญหาที่เกิดขึ้นจริง
  • Bthrow System.Exception ตรงๆ จากโค้ดของตัวเอง
  • Cใส่ข้อความอธิบายสาเหตุลงใน Message ก่อน throw
เฉลย: ข้อ B. ควรเลือก exception type ที่เจาะจงกับปัญหาเสมอ ไม่ throw คลาสฐานกว้างๆ อย่าง System.Exception ตรงๆ จากโค้ดตัวเอง

36.6 ⚠️ กับดักที่พบบ่อย

อาการสาเหตุ / วิธีแก้
แก้ ex.Message แล้ว error ว่า property เป็น read-only Message กำหนดได้แค่ตอนสร้าง object ผ่าน constructor เท่านั้น สร้าง exception object ใหม่ที่มีข้อความถูกต้องแทนการแก้ไขตัวเดิม
ใช้ throw ex; แทน throw; ตอน re-throw ทั้งสองแบบ compile ผ่าน แต่ throw ex; จะรีเซ็ต StackTrace ให้เริ่มนับ จากบรรทัดนั้นใหม่ ทำให้เสียร่องรอยต้นตอจริงของปัญหา — ใช้ throw; เปล่าๆ เพื่อรักษา StackTrace เดิมไว้เสมอ
throw ArgumentException ทั่วไปในทุกกรณี ไม่เจาะจง เลือกชนิดที่ตรงกับปัญหาที่สุด (ArgumentOutOfRangeException, ArgumentNullException ฯลฯ) จะช่วยให้ฝั่ง catch แยกแยะและรับมือได้แม่นยำกว่า
เขียนเงื่อนไข throw exception เพื่อควบคุม flow ปกติของโปรแกรม (เช่น throw แทนการ break ออกจาก loop) exception มีต้นทุนด้าน performance และทำให้โค้ดอ่านยาก — สงวนไว้สำหรับ เงื่อนไขผิดปกติจริงๆ เท่านั้น ใช้ if/else หรือ break/continue สำหรับ flow ปกติ

static void CheckAge(int age)
{

}

Create and Throw Exceptions Challenge (ดัดแปลงจาก Microsoft Learn) — จากโค้ด ต่อไปนี้ ที่ Process1 คืนค่าเป็น string บอกสถานะแทนการ throw จริง ให้ แปลงทั้งสามเมธอดเป็น void แล้วทำตามเงื่อนไข: Process1 ต้อง throw exception ตามชนิดปัญหาที่เจอ, Workflow1 ต้อง catch FormatException, top-level statements ต้อง catch DivideByZeroException

Program.cs (ตั้งต้น)
static string Process1(string[] userEntries)
{
    string returnMessage = "";
    foreach (string userValue in userEntries)
    {
        bool ok = int.TryParse(userValue, out int valueEntered);
        if (!ok)
        {
            returnMessage = "ข้อมูลไม่ถูกต้อง ต้องเป็นจำนวนเต็ม";
        }
        else if (valueEntered == 0)
        {
            returnMessage = "ข้อมูลไม่ถูกต้อง ห้ามเป็นศูนย์";
        }
    }
    return returnMessage == "" ? "process complete" : returnMessage;
}
💡 คำใบ้

แทนที่บรรทัด returnMessage = "..."; ทั้งสองจุด ด้วย throw new FormatException("..."); และ throw new DivideByZeroException("..."); ตามลำดับ

ดูเฉลย
Program.cs (เฉลย)
try
{
    Workflow1();
    Console.WriteLine("Workflow1 สำเร็จ");
}

static void Workflow1()
{
    string[] userEntries = { "1", "0", "2" };
    try
    {
        Process1(userEntries);
        Console.WriteLine("Process1 สำเร็จ");
    }
    catch (FormatException ex)
    {
        Console.WriteLine("Process1 พบปัญหา ยกเลิกการทำงาน");
        Console.WriteLine(ex.Message);
    }
}

static void Process1(string[] userEntries)
{
    foreach (string userValue in userEntries)
    {
        bool ok = int.TryParse(userValue, out int valueEntered);
        if (!ok)
        {
            throw new FormatException("ข้อมูลไม่ถูกต้อง ต้องเป็นจำนวนเต็ม");
        }
        if (valueEntered == 0)
        {
            throw new DivideByZeroException("ข้อมูลไม่ถูกต้อง ห้ามเป็นศูนย์");
        }
    }
}
ข้อ 3ยาก

อธิบายด้วยคำพูดตัวเอง (2-3 ประโยค) ว่าทำไม throw; ถึงดีกว่า throw ex; ตอน re-throw exception ในสถานการณ์ที่ต้องการรักษา StackTrace เดิมไว้

ดูแนวคำตอบ

แนวคำตอบตัวอย่าง: throw; ส่งต่อ exception object ตัวเดิมไปทั้งชุด รวมถึง StackTrace ที่ CLR สร้างไว้ตั้งแต่จุดเกิด exception จริง ทำให้ยังตามรอย ต้นตอปัญหาได้ครบ ส่วน throw ex; แม้จะโยน exception ชนิดเดียวกันออกไป แต่ C# จะรีเซ็ต StackTrace ให้เริ่มนับจากบรรทัด throw ex; นั้นแทน ทำให้ข้อมูลตำแหน่งจริงที่ทำให้เกิดปัญหาหายไป ยากต่อการ debug ทีหลัง

📌 สรุปบทที่ 36

บทนี้สอนให้คุณเป็นฝ่าย "โยน" exception เอง — สร้าง exception object ด้วย new ปรับแต่ง Message ตอนสร้าง แล้ว throw ด้วย throw คุณเรียนรู้ วิธีเลือกชนิด exception ที่เจาะจงที่สุดสำหรับปัญหาแต่ละแบบ วิธี re-throw exception เดิม ด้วย throw; เพื่อรักษา StackTrace ไว้ครบ และข้อควรเลี่ยงสำคัญๆ เช่น ห้ามใช้ exception แทนตรรกะควบคุม flow ปกติ

ตอนนี้คุณจับ (catch) และโยน (throw) exception เองได้ครบวงจรแล้ว บทถัดไปจะรวมทักษะทั้งหมด ของ Part 6 เข้าด้วยกันในโปรเจกต์แบบมีไกด์: ทั้ง debug ด้วย VS Code และเขียน exception handling ในแอปเดียวกัน
